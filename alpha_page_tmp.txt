// ── FETCH HELPERS ────────────────────────────────────────────────────────────
async function fetchJSON(url) {
  const res = await fetch(url, { cache: 'no-store' });
  if (!res.ok) throw new Error(`HTTP ${res.status} for ${url}`);
  const text = await res.text();
  // Fix NaN values — JSON spec doesn't support NaN
  const cleaned = text.replace(/:\s*NaN/g, ': null');
  return JSON.parse(cleaned);
}

async function loadAllData() {
  setLastUpdated('Refreshing...');
  try {
    const [ch, ac, cl, sp] = await Promise.allSettled([
      fetchJSON(URLS.challenge),
      fetchJSON(URLS.active),
      fetchJSON(URLS.closed),
      fetchJSON(URLS.strategy)
    ]);

    if (ch.status === 'fulfilled') challengeData = ch.value;
    activePicks = (ac.status === 'fulfilled' && Array.isArray(ac.value)) ? ac.value : [];
    closedPicks = (cl.status === 'fulfilled' && Array.isArray(cl.value)) ? cl.value : [];
    strategyPerf = (sp.status === 'fulfilled' && sp.value && typeof sp.value === 'object') ? sp.value : {};

    const now = new Date().toLocaleTimeString('en-US', { hour12: false });
    setLastUpdated(now);
    document.getElementById('footer-updated').textContent = new Date().toLocaleString();

    renderAll();
    resetRefreshTimer();

  } catch(e) {
    setLastUpdated('Error: ' + e.message);
    showErrorState(e.message);
  }
}

function updateChallengeTimer() {
  const el = document.getElementById('challenge-countdown');
  const sub = document.getElementById('challenge-status');
  if (!challengeEndTime) {
    el.textContent = '--:--:--';
    sub.textContent = 'No end time';
    return;
  }
  const now = new Date();
  const diff = challengeEndTime - now;
  if (diff <= 0) {
    el.textContent = 'EXPIRED';
    el.className = 'challenge-timer expired';
    sub.textContent = 'Challenge ended';
    return;
  }
  const h = Math.floor(diff / 3600000);
  const m = Math.floor((diff % 3600000) / 60000);
  const s = Math.floor((diff % 60000) / 1000);
  el.textContent = `${pad(h)}:${pad(m)}:${pad(s)}`;
  el.className = 'challenge-timer' + (diff < 600000 ? ' ending' : '');
  sub.textContent = 'Until challenge ends';
}

function pad(n) { return String(n).padStart(2, '0'); }

// ── CHART ─────────────────────────────────────────────────────────────────────
function renderChart() {
  const container = document.getElementById('chart-container');
  const badge = document.getElementById('chart-checks-badge');

  if (!challengeData || !challengeData.checks || challengeData.checks.length < 2) {
    container.innerHTML = '<div class="status-box">Not enough check data to draw chart (need ≥ 2 checks)</div>';
    badge.textContent = (challengeData?.checks?.length || 0) + ' checks';
    return;
  }

  const checks = challengeData.checks;
  badge.textContent = checks.length + ' checks';

  // Build series: sum of valid pnl per check
  const points = checks.map(c => {
    const scores = c.strategy_scores || [];
    const pnl = scores.reduce((s, x) => s + (isFinite(x.total_pnl_dollar) ? x.total_pnl_dollar : 0), 0);
    return { x: c.elapsed_minutes, pnl, wr: c.win_rate || 0, time: c.check_time };
  });

  const W = 700, H = 200;
  const padL = 58, padR = 20, padT = 16, padB = 36;
  const cW = W - padL - padR;
  const cH = H - padT - padB;

  const xs = points.map(p => p.x);
  const ys = points.map(p => p.pnl);
  const minX = Math.min(...xs), maxX = Math.max(...xs);
  const minY = Math.min(0, Math.min(...ys));
  const maxY = Math.max(0, Math.max(...ys));
  const rangeX = maxX - minX || 1;
  const rangeY = maxY - minY || 1;

  function sx(x) { return padL + ((x - minX) / rangeX) * cW; }
  function sy(y) { return padT + cH - ((y - minY) / rangeY) * cH; }

  // Y axis grid lines
  const yTicks = 4;
  let gridLines = '';
  let yLabels = '';
  for (let i = 0; i <= yTicks; i++) {
    const yVal = minY + (rangeY * i / yTicks);
    const yPx = sy(yVal);
    const cls = Math.abs(yVal) < 0.01 ? 'chart-zero-line' : 'chart-grid-line';
    gridLines += `<line x1="${padL}" x2="${W - padR}" y1="${yPx}" y2="${yPx}" class="${cls}"/>`;
    yLabels += `<text x="${padL - 6}" y="${yPx + 4}" class="chart-axis-label" text-anchor="end">${formatPnlShort(yVal)}</text>`;
  }

  // X axis labels
  let xLabels = '';
  points.forEach((p, i) => {
    if (i === 0 || i === points.length - 1 || points.length <= 6) {
      const xPx = sx(p.x);
      xLabels += `<text x="${xPx}" y="${H - 4}" class="chart-axis-label" text-anchor="middle">${formatElapsed(p.x)}</text>`;
    }
  });

  // Line path
  const positive = ys[ys.length - 1] >= 0;
  const lineColor = positive ? 'var(--green)' : 'var(--red)';
  const fillColor = positive ? 'rgba(34,197,94,0.12)' : 'rgba(239,68,68,0.12)';

  let pathD = '';
  let areaD = `M ${sx(points[0].x)} ${sy(0)} `;
  points.forEach((p, i) => {
    const px = sx(p.x), py = sy(p.pnl);
    if (i === 0) {
      pathD += `M ${px} ${py}`;
      areaD += `L ${px} ${py} `;
    } else {
      // Smooth curve
      const prev = points[i-1];
      const mx = (sx(prev.x) + px) / 2;
      pathD += ` C ${mx} ${sy(prev.pnl)}, ${mx} ${py}, ${px} ${py}`;
      areaD += `L ${px} ${py} `;
    }
  });
  areaD += `L ${sx(points[points.length-1].x)} ${sy(0)} Z`;

  // Dots with tooltips
  let dots = '';
  points.forEach(p => {
    const px = sx(p.x), py = sy(p.pnl);
    const tip = `${formatElapsed(p.x)}: ${formatPnl(p.pnl)} | WR: ${p.wr.toFixed(1)}%`;
    dots += `<circle cx="${px}" cy="${py}" r="4" fill="${lineColor}" stroke="var(--bg)" stroke-width="2" opacity="0.9"><title>${escHtml(tip)}</title></circle>`;
  });

  // Zero line
  const zeroY = sy(0);
  const zeroLine = `<line x1="${padL}" x2="${W - padR}" y1="${zeroY}" y2="${zeroY}" stroke="rgba(148,163,184,0.3)" stroke-width="1" stroke-dasharray="4 4"/>`;

  const svg = `<svg class="pnl-chart" viewBox="0 0 ${W} ${H}" preserveAspectRatio="xMidYMid meet">
    <defs>
      <linearGradient id="lineGrad" x1="0%" y1="0%" x2="0%" y2="100%">
        <stop offset="0%" stop-color="${lineColor}" stop-opacity="0.3"/>
        <stop offset="100%" stop-color="${lineColor}" stop-opacity="0.02"/>
      </linearGradient>
    </defs>
    ${gridLines}
    ${zeroLine}
    ${yLabels}
    ${xLabels}
    <path d="${areaD}" fill="url(#lineGrad)"/>
    <path d="${pathD}" fill="none" stroke="${lineColor}" stroke-width="2.5" stroke-linejoin="round" stroke-linecap="round"/>
    ${dots}
  </svg>`;

  container.innerHTML = svg;
}

// ── LEADERBOARD ───────────────────────────────────────────────────────────────
function renderLeaderboard() {
  const container = document.getElementById('leaderboard-container');
  const badge = document.getElementById('leaderboard-badge');

  if (!challengeData || !challengeData.checks || challengeData.checks.length === 0) {
    container.innerHTML = '<div class="status-box">No check data available</div>';
    return;
  }

  const checks = challengeData.checks;
  const lastCheck = checks[checks.length - 1];
  badge.textContent = `Check at ${formatElapsed(lastCheck.elapsed_minutes)}`;

  const scores = (lastCheck.strategy_scores || [])
    .filter(s => isFinite(s.total_pnl_dollar))
    .sort((a, b) => b.total_pnl_dollar - a.total_pnl_dollar);

  const noDataScores = (lastCheck.strategy_scores || []).filter(s => !isFinite(s.total_pnl_dollar));

  if (scores.length === 0 && noDataScores.length === 0) {
    container.innerHTML = '<div class="status-box">No strategy data in latest check</div>';
    return;
  }

  const allScores = [...scores, ...noDataScores];
  const maxAbsPnl = Math.max(...scores.map(s => Math.abs(s.total_pnl_dollar)), 1);

  const medals = ['🥇', '🥈', '🥉'];

  let rows = '';
  allScores.forEach((s, i) => {
    const rank = i + 1;
    const isPositive = isFinite(s.total_pnl_dollar) && s.total_pnl_dollar > 0;
    const isNegative = isFinite(s.total_pnl_dollar) && s.total_pnl_dollar < 0;
    const rowClass = isPositive ? 'row-positive' : (isNegative ? 'row-negative' : '');

    const rankCell = rank <= 3 && isFinite(s.total_pnl_dollar)
      ? `<span class="rank-medal">${medals[rank-1]}</span>`
      : `<span class="rank-num">${rank}</span>`;

    const pnlClass = isPositive ? 'pnl-positive right' : (isNegative ? 'pnl-negative right' : 'pnl-neutral right');
    const pnlText = isFinite(s.total_pnl_dollar) ? formatPnl(s.total_pnl_dollar) : 'N/A';
    const pnlPct = isFinite(s.avg_pnl_pct) ? ((s.avg_pnl_pct * 100).toFixed(2) + '%') : 'N/A';

    const barW = isFinite(s.total_pnl_dollar) ? Math.max(2, Math.round(Math.abs(s.total_pnl_dollar) / maxAbsPnl * 48)) : 0;
    const barClass = isNegative ? 'negative' : '';

    rows += `<tr class="${rowClass}">
      <td>${rankCell}</td>
      <td><span class="strategy-name">${escHtml(s.strategy)}</span></td>
      <td class="right mono">${s.picks}</td>
      <td class="right mono" style="color:var(--green)">${s.winning}</td>
      <td class="right mono" style="color:var(--red)">${s.losing}</td>
      <td class="right mono" style="color:var(--blue)">${s.tp_hits}</td>
      <td class="right mono" style="color:var(--orange)">${s.sl_hits}</td>
      <td class="right mono">${pnlPct}</td>
      <td class="right ${pnlClass}">${pnlText}<span class="mini-bar ${barClass}" style="width:${barW}px"></span></td>
    </tr>`;
  });

  container.innerHTML = `<div class="table-wrap">
    <table>
      <thead>
        <tr>
          <th>Rank</th>
          <th>Strategy</th>
          <th class="right">Picks</th>
          <th class="right">W</th>
          <th class="right">L</th>
          <th class="right">TP</th>
          <th class="right">SL</th>
          <th class="right">Avg PnL%</th>
          <th class="right">P&amp;L $</th>
        </tr>
      </thead>
      <tbody>${rows}</tbody>
    </table>
  </div>`;
}

// ── FORMATTERS ────────────────────────────────────────────────────────────────
function formatPnl(v) {
  if (!isFinite(v)) return 'N/A';
  const sign = v >= 0 ? '+' : '';
  return sign + '$' + Math.abs(v).toFixed(2);
}

function formatPnlShort(v) {
  if (!isFinite(v)) return '';
  if (Math.abs(v) >= 100) return (v >= 0 ? '+' : '') + '$' + v.toFixed(0);
  return (v >= 0 ? '+' : '') + '$' + v.toFixed(1);
}

function formatPrice(v) {
  if (v === null || v === undefined || !isFinite(v)) return 'N/A';
  if (v < 0.0001) return v.toExponential(4);
  if (v < 0.01) return v.toFixed(6);
  if (v < 1) return v.toFixed(5);
  if (v < 10) return v.toFixed(4);
  if (v < 100) return v.toFixed(3);
  if (v < 10000) return v.toFixed(2);
  return v.toLocaleString('en-US', { maximumFractionDigits: 0 });
}

function formatElapsed(min) {
  if (!isFinite(min)) return '?';
  if (min < 1) return '<1m';
  if (min < 60) return Math.round(min) + 'm';
  const h = Math.floor(min / 60);
  const m = Math.round(min % 60);
  return m > 0 ? `${h}h${m}m` : `${h}h`;
}

function escHtml(str) {
  if (str === null || str === undefined) return '';
  return String(str)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;');
}

// ── REFRESH TIMER ─────────────────────────────────────────────────────────────
function resetRefreshTimer() {
  refreshTimer = REFRESH_INTERVAL;
}

function tickRefresh() {
  refreshTimer--;
  if (refreshTimer <= 0) {
    loadAllData();
    return;
  }
  document.getElementById('refresh-countdown').textContent = refreshTimer;
  const pct = (refreshTimer / REFRESH_INTERVAL) * 100;
  document.getElementById('refresh-bar-fill').style.width = pct + '%';
}

  setInterval(tickRefresh, 1000);
