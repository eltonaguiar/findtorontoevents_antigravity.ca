def calculate_pl(side, entry_price, exit_price, quantity, leverage=10):
    """
    Calculate P/L for BTCUSD.V linear perpetual futures

    Args:
        side: 'Buy' (long) or 'Sell' (short)
        entry_price: Entry price in USD
        exit_price: Exit price in USD
        quantity: Position size in BTC
        leverage: Leverage multiplier (default 10)

    Returns:
        P/L in USD
    """
    if side == 'Buy':
        return (exit_price - entry_price) * quantity * leverage
    else:  # Sell
        return (entry_price - exit_price) * quantity * leverage
